# DevForge Production Multi-Language Environment
FROM node:20-bookworm-slim

# Install system compilers and runtimes for all 5 languages (Python, GCC, G++, Java)
RUN apt-get update && apt-get install -y --no-install-recommends \
    python3 \
    python3-pip \
    build-essential \
    gcc \
    g++ \
    default-jdk \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /app

# Copy dependency manifests
COPY package*.json ./

# Install npm dependencies
RUN npm install

# Copy application source code
COPY . .

# Build Vite frontend static bundle into dist/
RUN npm run build

# Default environment port
ENV PORT=5000
EXPOSE 5000

# Start unified production server
CMD ["node", "server/server.js"]
